idf_component_get_property(bt_dir bt COMPONENT_DIR)
idf_component_get_property(bt_lib bt COMPONENT_LIB)
set(conference_patch "${CMAKE_CURRENT_LIST_DIR}/patch.py")
set(conference_output "${CMAKE_BINARY_DIR}/dashbridge_conference")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${conference_patch}")
execute_process(COMMAND "${PYTHON}" "${conference_patch}" "${bt_dir}" "${conference_output}"
    RESULT_VARIABLE conference_result)
if(NOT conference_result EQUAL 0)
    message(FATAL_ERROR "Failed to prepare pinned HFP conference callback")
endif()
get_target_property(conference_sources ${bt_lib} SOURCES)
set(conference_updated "")
set(conference_replaced 0)
foreach(item IN LISTS conference_sources)
    if(item MATCHES "(^|/)hf_ag/btc_hf_ag\\.c$")
        list(APPEND conference_updated "${conference_output}/btc_hf_ag.c")
        math(EXPR conference_replaced "${conference_replaced} + 1")
    else()
        list(APPEND conference_updated "${item}")
    endif()
endforeach()
if(NOT conference_replaced EQUAL 1)
    message(FATAL_ERROR "Expected the pinned HFP AG source")
endif()
set_property(TARGET ${bt_lib} PROPERTY SOURCES "${conference_updated}")
