idf_component_get_property(bt_dir bt COMPONENT_DIR)
idf_component_get_property(bt_lib bt COMPONENT_LIB)
set(avrc_patch "${CMAKE_CURRENT_LIST_DIR}/patch.py")
set(avrc_output "${CMAKE_BINARY_DIR}/dashbridge_avrc")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${avrc_patch}")
execute_process(COMMAND "${PYTHON}" "${avrc_patch}" "${bt_dir}" "${avrc_output}"
    RESULT_VARIABLE avrc_result)
if(NOT avrc_result EQUAL 0)
    message(FATAL_ERROR "Failed to prepare pinned AVRCP media relay support")
endif()
get_target_property(avrc_sources ${bt_lib} SOURCES)
set(avrc_updated "")
set(avrc_replaced 0)
foreach(item IN LISTS avrc_sources)
    if(item MATCHES "(^|/)avrc/btc_avrc\\.c$")
        list(APPEND avrc_updated "${avrc_output}/btc_avrc.c")
        math(EXPR avrc_replaced "${avrc_replaced} + 1")
    else()
        list(APPEND avrc_updated "${item}")
    endif()
endforeach()
if(NOT avrc_replaced EQUAL 1)
    message(FATAL_ERROR "Expected the pinned AVRCP source")
endif()
set_property(TARGET ${bt_lib} PROPERTY SOURCES "${avrc_updated}")
