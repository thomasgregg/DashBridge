idf_component_get_property(bt_dir bt COMPONENT_DIR)
idf_component_get_property(bt_lib bt COMPONENT_LIB)
set(codec_patch "${CMAKE_CURRENT_LIST_DIR}/patch.py")
set(codec_output "${CMAKE_BINARY_DIR}/dashbridge_audio_codec")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${codec_patch}")
execute_process(COMMAND "${PYTHON}" "${codec_patch}" "${bt_dir}" "${codec_output}"
    RESULT_VARIABLE codec_result)
if(NOT codec_result EQUAL 0)
    message(FATAL_ERROR "Failed to prepare pinned HFP decoder correction")
endif()
get_target_property(codec_sources ${bt_lib} SOURCES)
set(codec_updated "")
set(codec_replaced 0)
foreach(item IN LISTS codec_sources)
    if(item MATCHES "(^|/)hf_ag/bta_ag_co\\.c$" OR
       item MATCHES "(^|/)hf_client/bta_hf_client_co\\.c$")
        get_filename_component(filename "${item}" NAME)
        list(APPEND codec_updated "${codec_output}/${filename}")
        math(EXPR codec_replaced "${codec_replaced} + 1")
    else()
        list(APPEND codec_updated "${item}")
    endif()
endforeach()
if(NOT codec_replaced EQUAL 2)
    message(FATAL_ERROR "Expected both pinned HFP codec sources")
endif()
set_property(TARGET ${bt_lib} PROPERTY SOURCES "${codec_updated}")
